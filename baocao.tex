% ── Slide 4: RAG vs GraphRAG ──────────────────────────────────────────────────
\begin{frame}[t]{4. RAG truyền thống vs.\ Graph-based RAG}
\small
\begin{columns}[T,onlytextwidth]
\begin{column}{0.34\textwidth}
  \begin{block}{RAG truyền thống}
  \scriptsize
  \begin{itemize}\setlength{\itemsep}{2pt}\setlength{\parskip}{0pt}
    \item Chunk $\to$ vector store
    \item Top-$k$ similarity search
    \item Không có quan hệ entity
    \item Thiếu trích dẫn rõ ràng
    \item Dễ triển khai, rẻ
  \end{itemize}
  \end{block}

  \vspace{0.25em}

  \begin{block}{GraphRAG (MS 2024)}
  \scriptsize
  \begin{itemize}\setlength{\itemsep}{2pt}\setlength{\parskip}{0pt}
    \item Đồ thị + community detection
    \item Global/local query mode
    \item Chi phí xây graph cao
    \item Không domain-specific
  \end{itemize}
  \end{block}

  \vspace{0.25em}

  \begin{block}{MedGraphRAG}
  \scriptsize
  \begin{itemize}\setlength{\itemsep}{2pt}\setlength{\parskip}{0pt}
    \item Đồ thị 3 tầng y khoa
    \item Tag-based summary
    \item U-Retrieval (top-down + bottom-up)
    \item Trích dẫn nguồn rõ ràng
  \end{itemize}
  \end{block}
\end{column}
\begin{column}{0.62\textwidth}
  \begin{block}{So sánh kết quả trên test sets — Accuracy (\%)}
  \scriptsize
  \setlength{\tabcolsep}{4pt}

      thay \texttt{seq\_ret()} (gọi LLM $N$ lần) bằng
